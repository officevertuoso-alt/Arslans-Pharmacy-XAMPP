SQL VERIFICATION QUERIES
------------------------------------------------------------
-- Check FEFO snapshot stored in sale_items
SELECT si.id, si.product_id, si.batch_id, si.quantity,
       si.batch_number, si.expiry_date, b.expiry_date AS b_expiry
FROM sale_items si
LEFT JOIN batches b ON b.id = si.batch_id
ORDER BY si.id DESC LIMIT 20;

-- Check multi-batch FEFO split for one sale
SELECT si.id, si.product_id, si.batch_id, si.batch_number,
       si.expiry_date, si.quantity, si.unit_price, si.total
FROM sale_items si
WHERE si.sale_id = <SALE_ID>
ORDER BY si.expiry_date ASC, si.id ASC;

-- Check partial payment credit record
SELECT s.invoice_no, s.payment_status, s.paid_amount,
       cc.total_amount AS credit_balance, cc.due_date, cc.status
FROM sales s
LEFT JOIN customer_credits cc ON cc.sale_id = s.id
WHERE s.payment_status IN ('partial','pending')
ORDER BY s.id DESC LIMIT 10;
